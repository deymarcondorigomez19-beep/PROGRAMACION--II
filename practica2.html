1. Clase Fracción (Sobrecarga de Operadores)

Implementación de la clase Fraccion ampliando métodos aritméticos, simplificación por MCD y conversión dinámica.

fraccion.pyPython
import math

class Fraccion:
    def __init__(self, a=0, b=1):
        self.__numerador = a
        self.__denominador = b

    def __add__(self, o):
        a = self.__numerador * o.__denominador + self.__denominador * o.__numerador
        b = self.__denominador * o.__denominador
        return Fraccion(a, b)

    def __sub__(self, o):
        a = self.__numerador * o.__denominador - self.__denominador * o.__numerador
        b = self.__denominador * o.__denominador
        return Fraccion(a, b)

    def __mul__(self, o):
        return Fraccion(self.__numerador * o.__numerador, self.__denominador * o.__denominador)

    def __truediv__(self, o):
        return Fraccion(self.__numerador * o.__denominador, self.__denominador * o.__numerador)

    def __eq__(self, o):
        if isinstance(o, Fraccion):
            f1 = self.simplifica()
            f2 = o.simplifica()
            return f1.__numerador == f2.__numerador and f1.__denominador == f2.__denominador
        return False

    def convertirADecimal(self):
        if self.__denominador == 0:
            raise ZeroDivisionError("División entre cero")
        return self.__numerador / self.__denominador

    def esInverso(self, o):
        prod = self * o
        return abs(prod.convertirADecimal() - 1.0) < 1e-9

    @staticmethod
    def parseFraccion(cadena):
        p = cadena.strip().split('/')
        return Fraccion(int(p[0]), int(p[1]) if len(p) > 1 else 1)

    def simplifica(self):
        mcd = math.gcd(self.__numerador, self.__denominador)
        return Fraccion(self.__numerador // mcd, self.__denominador // mcd)

    def __str__(self):
        return "{}/{}".format(self.__numerador, self.__denominador)

2. Clase MiPunto (Sobrecarga de Métodos)

Cálculo de distancias euclidianas mediante sobrecarga para objetos de tipo MiPunto o pares ordenados (x, y).

mipunto.pyPython
import math

class MiPunto:
    def __init__(self, x=0.0, y=0.0):
        self.__x = float(x)
        self.__y = float(y)

    def get_x(self): return self.__x
    def get_y(self): return self.__y

    def distancia(self, *args):
        if len(args) == 1 and isinstance(args[0], MiPunto):
            o = args[0]
            return math.sqrt((self.__x - o.get_x())**2 + (self.__y - o.get_y())**2)
        elif len(args) == 2:
            x2, y2 = float(args[0]), float(args[1])
            return math.sqrt((self.__x - x2)**2 + (self.__y - y2)**2)

    def __str__(self):
        return "({}, {})".format(self.__x, self.__y)

3. Álgebra Vectorial (Perpendicularidad y Paralelismo)

Verificación de propiedades vectoriales en 2D aplicando sobrecarga de métodos estáticos.

algebra_vectorial.pyPython
import math

class Vector2D:
    def __init__(self, x=0.0, y=0.0):
        self.x = float(x)
        self.y = float(y)

    def magnitud(self): return math.sqrt(self.x**2 + self.y**2)
    def dot(self, o): return self.x * o.x + self.y * o.y
    def cross(self, o): return self.x * o.y - self.y * o.x
    def __add__(self, o): return Vector2D(self.x + o.x, self.y + o.y)
    def __sub__(self, o): return Vector2D(self.x - o.x, self.y - o.y)
    def __mul__(self, r): return Vector2D(self.x * r, self.y * r)

class AlgebraVectorial:
    @staticmethod
    def perpendicular(a, b, criterio=1):
        if criterio == 1: return abs((a + b).magnitud() - (a - b).magnitud()) < 1e-9
        elif criterio == 3: return abs(a.dot(b)) < 1e-9

    @staticmethod
    def proyeccion_de_a_sobre_b(a, b):
        return b * (a.dot(b) / (b.magnitud()**2))

    @staticmethod
    def componente_de_a_en_b(a, b):
        return a.dot(b) / b.magnitud()

4. Vector Tridimensional (Sobrecarga de Operadores 3D)

Uso de operadores especiales (`+`, `*`, `^`, `/`) para productos escalares, vectoriales y normalización en $\mathbb{R}^3$.

vector3d.pyPython
import math

class Vector3D:
    def __init__(self, x=0.0, y=0.0, z=0.0):
        self.__x, self.__y, self.__z = float(x), float(y), float(z)

    def __add__(self, o):
        return Vector3D(self.__x + o.__x, self.__y + o.__y, self.__z + o.__z)

    def __mul__(self, o):
        if isinstance(o, (int, float)):
            return Vector3D(self.__x * o, self.__y * o, self.__z * o)
        return self.__x * o.__x + self.__y * o.__y + self.__z * o.__z

    def __xor__(self, o):
        return Vector3D(
            self.__y * o.__z - self.__z * o.__y,
            self.__z * o.__x - self.__x * o.__z,
            self.__x * o.__y - self.__y * o.__x
        )

    def magnitud(self):
        return math.sqrt(self.__x**2 + self.__y**2 + self.__z**2)

    def __truediv__(self, r):
        return Vector3D(self.__x / r, self.__y / r, self.__z / r)

    def normal(self):
        return self / self.magnitud()

    def __str__(self):
        return "({}, {}, {})".format(self.__x, self.__y, self.__z)
